Como as Traduções Funcionam no Código
// Dicionário de traduções:
const traducoes = {
  "pt": {
    saudacao: "Olá, mundo! 👋",
    nome: "O teu nome",
    enviar: "Enviar ✉️",
  },
  "en": { saudacao: "Hello, world! 👋", ... },
  "de": { saudacao: "Hallo, Welt! 👋", ... },
  "ja": { saudacao: "こんにちは！👋", ... },
  "ar": { saudacao: "مرحبا بالعالم! 👋", ... },
};

// Aplicar o idioma escolhido:
function mudarIdioma(codigo) {
  const t = traducoes[codigo];
  saudacao.texto = t.saudacao;
  botaoEnviar.texto = t.enviar;
}
